# 1. Inicializás Git en tu carpeta local
git init

# 2. Vinculás directamente con el repositorio de tu Casino
git remote add origin https://github.com/FIdelrodri/Casino_rayen.git

# 3. Te parás en la rama principal
git branch -M main

# 4. EL TRUCO: Acá te va a saltar la ventana flotante para iniciar sesión en el navegador
git fetch

# 5. Una vez logueado, te bajás todos los archivos del casino a tu compu
git pull origin main
